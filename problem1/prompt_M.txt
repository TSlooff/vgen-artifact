// This module implements a 16-bit Linear Feedback Shift Register
module lfsr_16bit(
    input clk,
    input reset,
    output reg [15:0] Q
    ); 
// This module can be used for random number generation
// Inputs:
// clk: the clock
// reset: the reset
// Outputs:
// Q: the random number
// At each clock cycle Q is updated
// When reset is high Q is set to 16'h1
// When reset is low Q is shifted and updated using linear feedback
