// This module implements the AES Substitution layer
module Slayer(
    input [127:0] Inp,
    output [127:0] Oup
    ); 
// Inp is the 128-bit input vector
// Oup is the 128-bit output of the substitution layer
// Each 8-bit chunk of the Inp vector is passed to the s-box module to get the corresponding 8-bit chunk of the Oup vector
// Which means  for(i=0;i<16;i++) Oup[8*i+7 : 4*i] = Sbox[ Inp [8*i+7 : 8*i] ]
